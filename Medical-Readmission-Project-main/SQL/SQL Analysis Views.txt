-- Medicare FFS national readmission trend analysis views

### Year-over-Year Change in Readmission Rate

`
CREATE ALGORITHM = UNDEFINED
DEFINER = `root`@`localhost`
SQL SECURITY DEFINER
VIEW `yoy_change` AS
SELECT
    `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date` AS `Service_Month_Date`,
    `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate` AS `Readmission_Rate`,
    (
        `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate`
        - LAG(
            `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate`,
            12
          ) OVER (
            ORDER BY `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`
          )
    ) AS `YoY_Change`
FROM `ffs-medicare-30-day-readmission-rate-puf`
ORDER BY `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`;

### Total Avoided Readmissions (Baseline Comparison)

CREATE ALGORITHM = UNDEFINED
DEFINER = `root`@`localhost`
SQL SECURITY DEFINER
VIEW `total_avoided_readmits` AS
SELECT
    SUM(
        (
            (
                (`ffs-medicare-30-day-readmission-rate-puf`.`Index_Stays`
                 * 19.024166666666662) / 100
            )
            - `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Stays`
        )
    ) AS `Total_Avoided_Readmissions`
FROM `ffs-medicare-30-day-readmission-rate-puf`;


### Seasonal Readmission Patterns

CREATE ALGORITHM = UNDEFINED
DEFINER = `root`@`localhost`
SQL SECURITY DEFINER
VIEW `seasonality` AS
SELECT
    `t`.`Month_Name` AS `Month_Name`,
    `t`.`Avg_Readmission_Rate` AS `Avg_Readmission_Rate`,
    (`t`.`Avg_Readmission_Rate` - `t`.`Overall_Avg`) AS `Diff_From_Overall`
FROM (
    SELECT
        MONTHNAME(
            `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`
        ) AS `Month_Name`,
        AVG(
            `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate`
        ) AS `Avg_Readmission_Rate`,
        (
            SELECT AVG(
                `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate`
            )
            FROM `ffs-medicare-30-day-readmission-rate-puf`
        ) AS `Overall_Avg`
    FROM `ffs-medicare-30-day-readmission-rate-puf`
    GROUP BY MONTHNAME(
        `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`
    )
) `t`
ORDER BY `t`.`Avg_Readmission_Rate` DESC;


### Rolling 3-Month Average Readmission Rate

CREATE ALGORITHM = UNDEFINED
DEFINER = `root`@`localhost`
SQL SECURITY DEFINER
VIEW `rolling_three_month_readmit_rate` AS
SELECT
    `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date` AS `Service_Month_Date`,
    `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate` AS `Readmission_Rate`,
    AVG(
        `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate`
    ) OVER (
        ORDER BY `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`

FROM `ffs-medicare-30-day-readmission-rate-puf`
ORDER BY `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`;

### HRRP Policy Period Comparison

CREATE ALGORITHM = UNDEFINED
DEFINER = `root`@`localhost`
SQL SECURITY DEFINER
VIEW `hrrp_periods` AS
SELECT
    `t`.`Policy_Period` AS `Policy_Period`,
    AVG(`t`.`Readmission_Rate`) AS `Avg_Readmission_Rate`,
    AVG(`t`.`Index_Stays`) AS `Avg_Index_Stays`
FROM (
    SELECT
        `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month`,
        `ffs-medicare-30-day-readmission-rate-puf`.`Index_Stays`,
        `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Stays`,
        `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate`,
        `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`,
        CASE
            WHEN `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date` < '2012-10-01'
            THEN 'Pre-HRRP'
            ELSE 'Post-HRRP'
        END AS `Policy_Period`
    FROM `ffs-medicare-30-day-readmission-rate-puf`
) `t`
GROUP BY `t`.`Policy_Period`;

### Avoided Readmissions by Month

CREATE ALGORITHM = UNDEFINED
DEFINER = `root`@`localhost`
SQL SECURITY DEFINER
VIEW `avoided_readmissions` AS
SELECT
    `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`,
    `ffs-medicare-30-day-readmission-rate-puf`.`Index_Stays`,
    `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate`,
    (
        (`ffs-medicare-30-day-readmission-rate-puf`.`Index_Stays`
         * 19.024166666666662) / 100
    ) AS `Expected_Readmits_At_Baseline`,
    `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Stays` AS `Actual_Readmits`,
    (
        (
            (`ffs-medicare-30-day-readmission-rate-puf`.`Index_Stays`
             * 19.024166666666662) / 100
        )
        - `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Stays`
    ) AS `Avoided_Readmissions`
FROM `ffs-medicare-30-day-readmission-rate-puf`
ORDER BY `ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`;


### Average Readmission Rate by Month

CREATE ALGORITHM = UNDEFINED
DEFINER = `root`@`localhost`
SQL SECURITY DEFINER
VIEW `average_monthly_readmit` AS
SELECT
    CASE `t`.`Month_Num`
        WHEN 1 THEN 'January'
        WHEN 2 THEN 'February'
        WHEN 3 THEN 'March'
        WHEN 4 THEN 'April'
        WHEN 5 THEN 'May'
        WHEN 6 THEN 'June'
        WHEN 7 THEN 'July'
        WHEN 8 THEN 'August'
        WHEN 9 THEN 'September'
        WHEN 10 THEN 'October'
        WHEN 11 THEN 'November'
        WHEN 12 THEN 'December'
    END AS `Service_Month_Name`,
    AVG(`t`.`Readmission_Rate`) AS `Avg_Readmission_Rate`
FROM (
    SELECT
        MONTH(`ffs-medicare-30-day-readmission-rate-puf`.`Service_Month_Date`) AS `Month_Num`,
        `ffs-medicare-30-day-readmission-rate-puf`.`Readmission_Rate`
    FROM `ffs-medicare-30-day-readmission-rate-puf`
) `t`
GROUP BY `t`.`Month_Num`

ORDER BY `t`.`Month_Num`;
